### Phase 2: Create UI Button System

**2.1 Button Helper Functions (Create_0.gml)**
```gml
// Button structure: { id, label, x, y, w, h, hotkey, action, enabled, hover }

create_button = function(_id, _label, _x, _y, _w, _h, _hotkey, _action) {
    return {
        id: _id,
        label: _label,
        x: _x, y: _y, w: _w, h: _h,
        hotkey: _hotkey,
        action: _action,
        enabled: true,
        hover: false
    };
};

draw_button = function(_btn) {
    var _col = _btn.hover ? c_yellow : (_btn.enabled ? c_white : c_gray);
    var _bg_col = _btn.hover ? c_dkgray : c_black;
    
    // Draw button background (rounded rect)
    draw_rectangle_color(_btn.x, _btn.y, _btn.x + _btn.w, _btn.y + _btn.h, 
                        _bg_col, _bg_col, _bg_col, _bg_col, false);
    
    // Draw border
    draw_rectangle(_btn.x, _btn.y, _btn.x + _btn.w, _btn.y + _btn.h, true);
    
    // Draw label
    draw_set_colour(_col);
    draw_set_halign(fa_center);
    draw_set_valign(fa_middle);
    draw_text(_btn.x + _btn.w/2, _btn.y + _btn.h/2, _btn.label);
    
    // Draw hotkey hint
    if (_btn.hotkey != "") {
        draw_set_colour(c_ltgray);
        draw_text(_btn.x + _btn.w/2, _btn.y + _btn.h + 16, "[" + _btn.hotkey + "]");
    }
};

is_button_hovered = function(_btn) {
    return mouse_x >= _btn.x && mouse_x <= _btn.x + _btn.w &&
           mouse_y >= _btn.y && mouse_y <= _btn.y + _btn.h;
};
```

**3.3 Input Handling (Step_0.gml)**
```gml
// Input cursor blink
input_cursor_timer++;
if (input_cursor_timer >= 30) {
    input_cursor_timer = 0;
    input_cursor_visible = !input_cursor_visible;
}

// Room code input
if (show_join_dialog) {
    // Handle character input
    if (keyboard_check_pressed(vk_anykey)) {
        var _key = keyboard_lastkey;
        if (_key >= ord("A") && _key <= ord("Z") && string_length(input_buffer) < input_max_length) {
            input_buffer += chr(_key);
        } else if (_key == vk_backspace && string_length(input_buffer) > 0) {
            input_buffer = string_delete(input_buffer, string_length(input_buffer), 1);
        } else if (_key == vk_enter && string_length(input_buffer) == 4) {
            obj_network_manager.join_room(input_buffer);
            show_join_dialog = false;
            input_buffer = "";
        } else if (_key == vk_escape) {
            show_join_dialog = false;
            input_buffer = "";
        }
    }
}
```

**6.4 Action Execution (Create_0.gml)**
```gml
execute_button_action = function(_action) {
    switch (_action) {
        case "create_room":
            obj_network_manager.create_room(2, obj_network_manager.my_name);
            break;
        case "open_join_dialog":
            show_join_dialog = true;
            input_buffer = "";
            break;
        case "toggle_ready":
            obj_network_manager.send_ready();
            break;
        case "leave_room":
            obj_network_manager.send_leave();
            break;
        case "rematch":
            // Reset and ready up for rematch
            obj_network_manager.send_ready(true);
            break;
        case "return_lobby":
            // Return to lobby state, wait for host to restart
            break;
    }
};
```

### Risks & Considerations

1. **Font Rendering:** Ensure `fnt_text` supports all characters needed (uppercase letters, numbers, symbols)
2. **Input Handling:** Room code input should accept only uppercase letters (A-Z) and limit to 4 characters
3. **State Synchronization:** UI state must always reflect `obj_network_manager.room_state` accurately
4. **Performance:** UI rendering should be efficient; avoid expensive operations in Draw events
5. **Accessibility:** Keep keyboard shortcuts functional alongside mouse interaction
6. **Room Transitions:** Ensure UI properly handles joining/leaving rooms and reconnection
7. **Button Layering:** Ensure UI draws above game elements (use Draw_GUI for overlays)
